// Standalone named-file support for the unchanged private Chrome fixture bridge.
// Exact process/directory/window functions are appended from their original
// source by stage-native-manual-chrome.mjs, with SHA-256 provenance. No Engine,
// Store, runtime dispatch or retained-profile implementation is imported.
import (
	"context"
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"time"

	"github.com/abangkis/AkuSidecar/internal/ownedprocess"

	"golang.org/x/sys/windows"
)

type processFixtureIdentity = ownedprocess.Identity
type processFixtureGrant struct {
	Nonce string
	Parent processFixtureIdentity
	Reopen, profilePrepared bool
}
type processFixtureStart struct { Grant processFixtureGrant; PolicyRecovery *struct{} }
type SyntheticSessionEvidence struct { Authenticated bool; AccountID, Revision string }

// synthetic harness. Unused bridge branches must fail rather than fake success.

	return errors.New("standalone-process-grant-unavailable")
}
func validateProcessFixtureStartScope(string, processFixtureStart) error {
	return errors.New("standalone-process-grant-unavailable")
}
